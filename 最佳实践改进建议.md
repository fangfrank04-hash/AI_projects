### 1.2 增强错误处理和重试机制

**新增工具函数**（`proposal-python-v2/utils/error_handler.py`）：
```python
"""
错误处理工具
提供重试、熔断、错误分类等功能
"""
import asyncio
from functools import wraps
from typing import Callable, Any

class APIError(Exception):
    """API错误基类"""
    def __init__(self, message: str, code: int = 500, details: dict = None):
        self.message = message
        self.code = code
        self.details = details or {}
        super().__init__(self.message)


def retry_on_failure(max_retries: int = 3, delay: float = 1.0):
    """
    重试装饰器
    用法：
    @retry_on_failure(max_retries=3, delay=1.0)
    async def my_func():
        ...
    """
    def decorator(func: Callable):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            last_exception = None
            for attempt in range(max_retries):
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    last_exception = e
                    if attempt < max_retries - 1:
                        wait_time = delay * (2 ** attempt)  # 指数退避
                        print(f"[Retry] {func.__name__} failed, retrying in {wait_time}s... (attempt {attempt + 1}/{max_retries})")
                        await asyncio.sleep(wait_time)
            raise last_exception
        return wrapper
    return decorator


async def handle_api_error(e: Exception) -> dict:
    """
    API错误处理器
    将异常转换为标准的错误响应
    """
    if isinstance(e, APIError):
        return {"success": False, "message": e.message, "code": e.code, "details": e.details}
    else:
        return {"success": False, "message": f"内部错误: {str(e)}", "code": 500}
```

### 3.1 添加工具调用日志和监控

**改进代码**（`proposal-python-v2/mcp_server/http_server.py`）：
```python
import time
from functools import wraps

# 工具调用统计
tool_stats = {
    "total_calls": 0,
    "success_calls": 0,
    "failed_calls": 0,
    "tool_counts": {}  # 每个工具的调用次数
}

def log_tool_call(func):
    """工具调用日志装饰器"""
    @wraps(func)
    async def wrapper(*args, **kwargs):
        global tool_stats
        tool_name = func.__name__
        
        # 初始化统计
        if tool_name not in tool_stats["tool_counts"]:
            tool_stats["tool_counts"][tool_name] = {"success": 0, "failed": 0}
        
        start_time = time.time()
        try:
            tool_stats["total_calls"] += 1
            result = await func(*args, **kwargs)
            elapsed = time.time() - start_time
            
            tool_stats["success_calls"] += 1
            tool_stats["tool_counts"][tool_name]["success"] += 1
            
            print(f"[MCP] Tool {tool_name} executed in {elapsed:.2f}s - SUCCESS")
            return result
        except Exception as e:
            elapsed = time.time() - start_time
            
            tool_stats["failed_calls"] += 1
            tool_stats["tool_counts"][tool_name]["failed"] += 1
            
            print(f"[MCP] Tool {tool_name} failed after {elapsed:.2f}s - ERROR: {e}")
            raise
    return wrapper


# 应用装饰器到所有工具
@mcp.tool()
@log_tool_call

    """查询项目基本信息"""
    # ... 原有逻辑
```

public class GlobalExceptionHandler {
    
    @ExceptionHandler(ResourceNotFoundException.class)

        ErrorResponse error = new ErrorResponse(
            "NOT_FOUND",
            e.getMessage(),
            Instant.now().toEpochMilli()
        );
        return ResponseEntity.status(404).body(error);
    }
    
    @ExceptionHandler(InvalidArgumentException.class)
    public ResponseEntity<ErrorResponse> handleInvalidArgument(InvalidArgumentException e) {
        ErrorResponse error = new ErrorResponse(
            "INVALID_ARGUMENT",
            e.getMessage(),
            Instant.now().toEpochMilli()
        );
        return ResponseEntity.status(400).body(error);
    }

            "INTERNAL_ERROR",
            "系统内部错误",
            Instant.now().toEpochMilli()
        );
        return ResponseEntity.status(500).body(error);
    }
}

## 六、监控和日志改进建议

### 6.1 集成Prometheus + Grafana

**Python端**（`proposal-python-v2/main.py`）：
```python
from prometheus_client import Counter, Histogram, generate_latest
from fastapi import Response

# 定义指标
REQUEST_COUNT = Counter("api_requests_total", "Total API requests", ["method", "endpoint"])
REQUEST_LATENCY = Histogram("api_request_duration_seconds", "Request latency", ["endpoint"])

@app.get("/metrics")
async def metrics():
    """Prometheus指标端点"""
    return Response(generate_latest(), media_type="text/plain")

@app.middleware("http")
async def add_metrics(request, call_next):
    """中间件：记录请求指标"""
    start_time = time.time()
    response = await call_next(request)
    process_time = time.time() - start_time
    
    REQUEST_COUNT.labels(request.method, request.url.path).inc()
    REQUEST_LATENCY.labels(request.url.path).observe(process_time)
    
    return response
```

### 6.2 结构化日志

**Python端**（`proposal-python-v2/utils/logging_config.py`）：
```python
"""
日志配置
输出JSON格式日志，方便ELK或Loki收集
"""
import json
import logging
from datetime import datetime

class JSONFormatter(logging.Formatter):
    """JSON日志格式化器"""
    def format(self, record):
        log_obj = {
            "timestamp": datetime.utcnow().isoformat(),
            "level": record.levelname,
            "logger": record.name,
            "message": record.getMessage(),
            "module": record.module,
            "line": record.lineno
        }
        
        if record.exc_info:
            log_obj["exception"] = self.formatException(record.exc_info)
        
        return json.dumps(log_obj, ensure_ascii=False)

# 配置根日志器
def setup_logging():
    handler = logging.StreamHandler()
    handler.setFormatter(JSONFormatter())
    
    root_logger = logging.getLogger()
    root_logger.setLevel(logging.INFO)
    root_logger.addHandler(handler)
```
