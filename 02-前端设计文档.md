# 前端设计文档 - 项目方案书AI自动填写系统

## 一、技术选型

| 技术项 | 选型 | 版本 | 说明 |
|--------|------|------|------|
| 框架 | React | ^18.2.0 | 已有项目使用，保持统一 |
| 构建工具 | Vite | ^5.0.0 | 快速冷启动，HMR |
| 状态管理 | React Context + useReducer | 内置 | 无需引入第三方库，减少依赖 |
| HTTP客户端 | fetch API | 内置 | 原生支持，无需axios |
| UI样式 | CSS Modules / Tailwind | - | 与现有项目保持一致 |
| 类型系统 | TypeScript | ^5.3.0 | 类型安全 |

**选型理由**：
- 与原有项目技术栈保持一致（React）
- 减少学习成本，直接复用现有组件和工具函数
- 不引入Pinia/Redux等额外状态管理库，用React内置能力足够

### 3.1 认证状态 (AuthContext)

```typescript
// contexts/AuthContext.tsx
import React, { createContext, useContext, useState, useCallback } from 'react';

interface UserInfo {
  userId: string;
  userName: string;
  dept: string;
  roles: string[];
}

interface AuthState {
  currentUser: UserInfo | null;
  token: string | null;
  isAuthenticated: boolean;
  loginExpired: boolean;
}

interface AuthContextType extends AuthState {
  login: (ssoToken: string) => Promise<void>;
  logout: () => void;
  checkAuth: () => boolean;
  setLoginExpired: (expired: boolean) => void;
}

const AuthContext = createContext<AuthContextType | null>(null);

export const AuthProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [state, setState] = useState<AuthState>({
    currentUser: null,
    token: localStorage.getItem('token'),
    isAuthenticated: !!localStorage.getItem('token'),
    loginExpired: false,
  });

  const login = useCallback(async (ssoToken: string) => {
    // 调用Java登录接口，换取JWT Token
    const response = await fetch('/api/auth/sso-login', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ ssoToken }),
    });
    const data = await response.json();
    localStorage.setItem('token', data.token);
    setState({
      currentUser: data.user,
      token: data.token,
      isAuthenticated: true,
      loginExpired: false,
    });
  }, []);

  const logout = useCallback(() => {
    localStorage.removeItem('token');
    setState({
      currentUser: null,
      token: null,
      isAuthenticated: false,
      loginExpired: false,
    });
  }, []);

  const checkAuth = useCallback(() => {
    return state.isAuthenticated && !state.loginExpired;
  }, [state.isAuthenticated, state.loginExpired]);

  const setLoginExpired = useCallback((expired: boolean) => {
    setState(prev => ({ ...prev, loginExpired: expired }));
  }, []);

  return (
    <AuthContext.Provider value={{ ...state, login, logout, checkAuth, setLoginExpired }}>
      {children}
    </AuthContext.Provider>
  );
};

export const useAuth = () => {
  const context = useContext(AuthContext);
  if (!context) throw new Error('useAuth必须在AuthProvider内使用');
  return context;
};
```

### 3.2 AI聊天状态 (AIChatContext) - 核心

```typescript
// contexts/AIChatContext.tsx
import React, { createContext, useContext, useReducer, useCallback } from 'react';

// ==================== 类型定义 ====================

export type StepNumber = 1 | 2 | 3 | 4 | 5;

export interface Message {
  id: string;
  role: 'ai' | 'user';
  content: string;
  type: 'text' | 'preview_team' | 'preview_control' | 'preview_schedule' | 'preview_resource' | 'preview_quality' | 'error' | 'complete';
  timestamp: number;
  data?: any; // 预览卡片的数据
}

export interface StepState {
  step: StepNumber;
  status: 'pending' | 'generating' | 'preview' | 'confirmed';
  data: any | null;
  draftData: any | null; // 用户编辑后的草稿
}

export interface AIChatState {
  isOpen: boolean;
  isMaximized: boolean;
  messages: Message[];
  currentStep: StepNumber;
  steps: Record<StepNumber, StepState>;
  isTyping: boolean;
  isGenerating: boolean;
  generationError: string | null;
  position: { x: number; y: number };
}

type AIChatAction =
  | { type: 'TOGGLE_CHAT' }
  | { type: 'TOGGLE_MAXIMIZE' }
  | { type: 'SET_POSITION'; payload: { x: number; y: number } }
  | { type: 'ADD_MESSAGE'; payload: Message }
  | { type: 'SET_TYPING'; payload: boolean }
  | { type: 'SET_GENERATING'; payload: boolean }
  | { type: 'SET_ERROR'; payload: string | null }
  | { type: 'UPDATE_STEP_STATUS'; payload: { step: StepNumber; status: StepState['status'] } }
  | { type: 'UPDATE_STEP_DATA'; payload: { step: StepNumber; data: any } }
  | { type: 'UPDATE_STEP_DRAFT'; payload: { step: StepNumber; draftData: any } }
  | { type: 'CONFIRM_STEP'; payload: StepNumber }
  | { type: 'RESET_STEP'; payload: StepNumber }
  | { type: 'RESET_ALL' };

const initialSteps: Record<StepNumber, StepState> = {
  1: { step: 1, status: 'pending', data: null, draftData: null },
  2: { step: 2, status: 'pending', data: null, draftData: null },
  3: { step: 3, status: 'pending', data: null, draftData: null },
  4: { step: 4, status: 'pending', data: null, draftData: null },
  5: { step: 5, status: 'pending', data: null, draftData: null },
};

const initialState: AIChatState = {
  isOpen: false,
  isMaximized: false,
  messages: [],
  currentStep: 1,
  steps: initialSteps,
  isTyping: false,
  isGenerating: false,
  generationError: null,
  position: { x: window.innerWidth - 420, y: 100 },
};

// ==================== Reducer ====================

function aiChatReducer(state: AIChatState, action: AIChatAction): AIChatState {
  switch (action.type) {
    case 'TOGGLE_CHAT':
      return { ...state, isOpen: !state.isOpen };
    case 'TOGGLE_MAXIMIZE':
      return { ...state, isMaximized: !state.isMaximized };
    case 'SET_POSITION':
      return { ...state, position: action.payload };
    case 'ADD_MESSAGE':
      return { ...state, messages: [...state.messages, action.payload] };
    case 'SET_TYPING':
      return { ...state, isTyping: action.payload };
    case 'SET_GENERATING':
      return { ...state, isGenerating: action.payload };
    case 'SET_ERROR':
      return { ...state, generationError: action.payload };
    case 'UPDATE_STEP_STATUS':
      return {
        ...state,
        steps: {
          ...state.steps,
          [action.payload.step]: {
            ...state.steps[action.payload.step],
            status: action.payload.status,
          },
        },
      };
    case 'UPDATE_STEP_DATA':
      return {
        ...state,
        steps: {
          ...state.steps,
          [action.payload.step]: {
            ...state.steps[action.payload.step],
            data: action.payload.data,
            draftData: action.payload.data,
          },
        },
      };
    case 'UPDATE_STEP_DRAFT':
      return {
        ...state,
        steps: {
          ...state.steps,
          [action.payload.step]: {
            ...state.steps[action.payload.step],
            draftData: action.payload.draftData,
          },
        },
      };
    case 'CONFIRM_STEP': {
      const step = action.payload;
      const nextStep = (step + 1) as StepNumber;
      return {
        ...state,
        steps: {
          ...state.steps,
          [step]: {
            ...state.steps[step],
            status: 'confirmed',
            data: state.steps[step].draftData,
          },
        },
        currentStep: nextStep > 5 ? 5 : nextStep,
      };
    }
    case 'RESET_STEP':
      return {
        ...state,
        steps: {
          ...state.steps,
          [action.payload]: {
            ...state.steps[action.payload],
            status: 'pending',
            draftData: null,
          },
        },
        currentStep: action.payload,
      };
    case 'RESET_ALL':
      return { ...initialState, isOpen: state.isOpen };
    default:
      return state;
  }
}

// ==================== Context ====================

interface AIChatContextType extends AIChatState {
  dispatch: React.Dispatch<AIChatAction>;
  generateStep: (step: StepNumber) => Promise<void>;
  confirmStep: (step: StepNumber) => Promise<void>;
  sendChatMessage: (message: string) => Promise<void>;
  jumpToStep: (step: StepNumber) => void;
}

const AIChatContext = createContext<AIChatContextType | null>(null);

export const AIChatProvider: React.FC<{ children: React.ReactNode; projectId: string }> = ({
  children,
  projectId,
}) => {
  const [state, dispatch] = useReducer(aiChatReducer, initialState);

  // 生成指定步骤
  const generateStep = useCallback(async (step: StepNumber) => {
    dispatch({ type: 'SET_GENERATING', payload: true });
    dispatch({ type: 'SET_ERROR', payload: null });
    dispatch({ type: 'UPDATE_STEP_STATUS', payload: { step, status: 'generating' } });

    try {
      const response = await fetch(`/api/project/${projectId}/ai/generate/${step}`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${localStorage.getItem('token')}`,
        },
      });

      if (!response.ok) {
        if (response.status === 401) {
          dispatch({ type: 'SET_ERROR', payload: '登录已过期，请重新登录' });
          return;
        }
        throw new Error('生成失败');
      }

      const data = await response.json();

      dispatch({ type: 'UPDATE_STEP_DATA', payload: { step, data: data.content } });
      dispatch({ type: 'UPDATE_STEP_STATUS', payload: { step, status: 'preview' } });
      dispatch({
        type: 'ADD_MESSAGE',
        payload: {
          id: Date.now().toString(),
          role: 'ai',
          content: getPreviewMessage(step),
          type: getPreviewType(step),
          timestamp: Date.now(),
          data: data.content,
        },
      });
    } catch (error) {
      dispatch({ type: 'SET_ERROR', payload: error instanceof Error ? error.message : '生成失败' });
    } finally {
      dispatch({ type: 'SET_GENERATING', payload: false });
    }
  }, [projectId]);

  // 确认步骤并回填
  const confirmStep = useCallback(async (step: StepNumber) => {
    const draftData = state.steps[step].draftData;
    if (!draftData) return;

    try {
      const response = await fetch(`/api/project/${projectId}/proposal`, {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${localStorage.getItem('token')}`,
        },
        body: JSON.stringify({ step, data: draftData }),
      });

      if (!response.ok) {
        if (response.status === 401) {
          dispatch({ type: 'SET_ERROR', payload: '登录已过期，请重新登录' });
          return;
        }
        throw new Error('回填失败');
      }

      dispatch({ type: 'CONFIRM_STEP', payload: step });

      // 如果还有下一步，自动开始生成
      const nextStep = (step + 1) as StepNumber;
      if (nextStep <= 5) {
        dispatch({
          type: 'ADD_MESSAGE',
          payload: {
            id: Date.now().toString(),
            role: 'ai',
            content: `步骤${step}已确认，正在生成步骤${nextStep}...`,
            type: 'text',
            timestamp: Date.now(),
          },
        });
        await generateStep(nextStep);
      } else {
        dispatch({
          type: 'ADD_MESSAGE',
          payload: {
            id: Date.now().toString(),
            role: 'ai',
            content: '🎉 恭喜！方案书全部5个模块已确认完成！所有数据已回填至主页面。',
            type: 'complete',
            timestamp: Date.now(),
          },
        });
      }
    } catch (error) {
      dispatch({ type: 'SET_ERROR', payload: error instanceof Error ? error.message : '回填失败' });
    }
  }, [projectId, state.steps, generateStep]);

  // 发送聊天消息（修改指令）
  const sendChatMessage = useCallback(async (message: string) => {
    dispatch({ type: 'SET_TYPING', payload: true });

    try {
      const response = await fetch(`/api/project/${projectId}/ai/chat`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${localStorage.getItem('token')}`,
        },
        body: JSON.stringify({
          message,
          currentStep: state.currentStep,
          draftData: state.steps[state.currentStep].draftData,
        }),
      });

      if (!response.ok) {
        if (response.status === 401) {
          dispatch({ type: 'SET_ERROR', payload: '登录已过期，请重新登录' });
          return;
        }
        throw new Error('对话失败');
      }

      const data = await response.json();

      // 如果返回了更新后的数据，更新草稿
      if (data.updatedData) {
        dispatch({
          type: 'UPDATE_STEP_DRAFT',
          payload: { step: state.currentStep, draftData: data.updatedData },
        });
      }

      dispatch({
        type: 'ADD_MESSAGE',
        payload: {
          id: Date.now().toString(),
          role: 'ai',
          content: data.message,
          type: 'text',
          timestamp: Date.now(),
        },
      });
    } catch (error) {
      dispatch({
        type: 'ADD_MESSAGE',
        payload: {
          id: Date.now().toString(),
          role: 'ai',
          content: '抱歉，处理您的指令时出错了，请重试。',
          type: 'error',
          timestamp: Date.now(),
        },
      });
    } finally {
      dispatch({ type: 'SET_TYPING', payload: false });
    }
  }, [projectId, state.currentStep, state.steps]);

  const jumpToStep = useCallback((step: StepNumber) => {
    dispatch({ type: 'RESET_STEP', payload: step });
  }, []);

  return (
    <AIChatContext.Provider
      value={{ ...state, dispatch, generateStep, confirmStep, sendChatMessage, jumpToStep }}
    >
      {children}
    </AIChatContext.Provider>
  );
};

export const useAIChat = () => {
  const context = useContext(AIChatContext);
  if (!context) throw new Error('useAIChat必须在AIChatProvider内使用');
  return context;
};

// ==================== 辅助函数 ====================

function getPreviewMessage(step: StepNumber): string {
  const messages: Record<StepNumber, string> = {
    1: '已生成项目团队职责分配，请确认或编辑后点击回填。',
    2: '已生成管控方案，请确认或编辑后点击回填。',
    3: '已生成项目进度计划，请确认或编辑后点击回填。',
    4: '已生成项目资源计划，请确认或编辑后点击回填。',
    5: '已生成质量保证计划，请确认或编辑后点击回填。',
  };
  return messages[step];
}

function getPreviewType(step: StepNumber): Message['type'] {
  const types: Record<StepNumber, Message['type']> = {
    1: 'preview_team',
    2: 'preview_control',
    3: 'preview_schedule',
    4: 'preview_resource',
    5: 'preview_quality',
  };
  return types[step];
}
```

### 4.2 AIChatbot主组件

```tsx
// components/AIChatbot/index.tsx
import React, { useEffect } from 'react';
import { createPortal } from 'react-dom';
import { useAIChat } from '../../contexts/AIChatContext';
import { useAuth } from '../../contexts/AuthContext';
import { useProject } from '../../hooks/useProject';
import { DraggableWrapper } from './DraggableWrapper';
import { ChatWindow } from './ChatWindow';
import { ChatFloatButton } from './ChatFloatButton';

interface AIChatbotProps {
  projectId: string;
}

export const AIChatbot: React.FC<AIChatbotProps> = ({ projectId }) => {
  const { isOpen, dispatch, generateStep, currentStep } = useAIChat();
  const { currentUser } = useAuth();
  const { project, teamMembers } = useProject(projectId);

  // 判断当前用户是否是项目经理
  const isPM = currentUser?.userName === project?.pmName;

  // 首次打开时，检查信息完整性并自动开始
  useEffect(() => {
    if (!isOpen || !isPM) return;

    // 检查项目信息和团队信息是否完整
    const hasProjectInfo = !!project?.name && !!project?.level;
    const hasTeamInfo = teamMembers && teamMembers.length > 0;

    if (!hasProjectInfo || !hasTeamInfo) {
      dispatch({
        type: 'ADD_MESSAGE',
        payload: {
          id: 'welcome',
          role: 'ai',
          content: hasTeamInfo
            ? '检测到项目信息尚未完善，请先完善项目信息后再开始方案书填写。'
            : '检测到项目团队信息尚未维护，请先维护项目团队后再开始方案书自动填写。',
          type: 'text',
          timestamp: Date.now(),
        },
      });
      return;
    }

    // 发送问候语并自动开始步骤1
    dispatch({
      type: 'ADD_MESSAGE',
      payload: {
        id: 'welcome',
        role: 'ai',
        content: `您好，${currentUser?.userName}！我是您的AI项目助手。\n✅ 检测到项目及团队信息已维护完整，即将开始【自动填写分模块确认流程】。`,
        type: 'text',
        timestamp: Date.now(),
      },
    });

    // 延迟后开始生成步骤1
    const timer = setTimeout(() => {
      generateStep(1);
    }, 1000);

    return () => clearTimeout(timer);
  }, [isOpen, isPM, project, teamMembers, currentUser, dispatch, generateStep]);

  // 非项目经理不显示AI助手
  if (!isPM) return null;

  return createPortal(
    <div className="ai-chatbot-root">
      {isOpen ? (
        <DraggableWrapper>
          <ChatWindow projectId={projectId} />
        </DraggableWrapper>
      ) : (
        <ChatFloatButton onClick={() => dispatch({ type: 'TOGGLE_CHAT' })} />
      )}
    </div>,
    document.body
  );
};
```

### 4.3 预览卡片组件（以TeamPreview为例）

```tsx
// components/AIChatbot/PreviewCards/TeamPreview.tsx
import React from 'react';
import { useAIChat, StepNumber } from '../../../contexts/AIChatContext';

interface TeamMember {
  role: string;
  name: string;
  responsibilities: string[];
}

interface TeamPreviewProps {
  data: TeamMember[];
  onConfirm: () => void;
}

export const TeamPreview: React.FC<TeamPreviewProps> = ({ data, onConfirm }) => {
  const { dispatch, currentStep } = useAIChat();

  const handleResponsibilityChange = (index: number, newResponsibilities: string) => {
    const updatedData = [...data];
    updatedData[index] = {
      ...updatedData[index],
      responsibilities: newResponsibilities.split('\n').filter(Boolean),
    };
    dispatch({
      type: 'UPDATE_STEP_DRAFT',
      payload: { step: currentStep as StepNumber, draftData: updatedData },
    });
  };

  return (
    <div className="preview-card team-preview">
      <div className="preview-header">
        <span className="step-badge">步骤 1/5</span>
        <h4>项目团队与职责确认</h4>
      </div>

      <table className="preview-table">
        <thead>
          <tr>
            <th>序号</th>
            <th>角色</th>
            <th>人员</th>
            <th>职责</th>
          </tr>
        </thead>
        <tbody>
          {data.map((member, index) => (
            <tr key={index}>
              <td>{index + 1}</td>
              <td>{member.role}</td>
              <td>{member.name}</td>
              <td>
                <textarea
                  value={member.responsibilities.join('\n')}
                  onChange={(e) => handleResponsibilityChange(index, e.target.value)}
                  rows={3}
                  className="responsibility-input"
                />
              </td>
            </tr>
```

```typescript
// utils/request.ts
const BASE_URL = import.meta.env.VITE_API_BASE_URL || '/api';

interface RequestOptions extends RequestInit {
  params?: Record<string, string>;
}

export async function request<T>(url: string, options: RequestOptions = {}): Promise<T> {
  const { params, ...rest } = options;

  // 构建URL（带查询参数）
  const queryString = params ? '?' + new URLSearchParams(params).toString() : '';
  const fullUrl = `${BASE_URL}${url}${queryString}`;

  // 自动添加Token
  const token = localStorage.getItem('token');
  const headers: Record<string, string> = {
    'Content-Type': 'application/json',
    ...((rest.headers as Record<string, string>) || {}),
  };
  if (token) {
    headers['Authorization'] = `Bearer ${token}`;
  }

  const response = await fetch(fullUrl, {
    ...rest,
    headers,
  });

  // 处理401
  if (response.status === 401) {
    localStorage.removeItem('token');
    window.dispatchEvent(new CustomEvent('login-expired'));
    throw new Error('登录已过期，请重新登录');
  }

  // 处理其他错误
  if (!response.ok) {
    const errorData = await response.json().catch(() => ({}));
    throw new Error(errorData.message || `请求失败: ${response.status}`);
  }

  return response.json();
}

// GET请求
export const get = <T>(url: string, params?: Record<string, string>) =>
  request<T>(url, { method: 'GET', params });

// POST请求
export const post = <T>(url: string, body: any) =>
  request<T>(url, { method: 'POST', body: JSON.stringify(body) });

// PUT请求
export const put = <T>(url: string, body: any) =>
  request<T>(url, { method: 'PUT', body: JSON.stringify(body) });
```

```typescript
// hooks/useAuth.ts
import { useEffect } from 'react';
import { useAuth as useAuthContext } from '../contexts/AuthContext';

export function useAuthGuard() {
  const { isAuthenticated, loginExpired, setLoginExpired } = useAuthContext();

  useEffect(() => {
    // 监听登录过期事件
    const handleLoginExpired = () => {
      setLoginExpired(true);
      alert('登录已过期，请重新登录');
      window.location.href = '/login';
    };

    window.addEventListener('login-expired', handleLoginExpired);
    return () => window.removeEventListener('login-expired', handleLoginExpired);
  }, [setLoginExpired]);

  return { isAuthenticated, loginExpired };
}
```

## 九、与Java/Python的协作约定

| 协作点 | 前端责任 | Java责任 | Python责任 |
|--------|----------|----------|------------|
| 数据获取 | 调用Java API获取项目/团队数据 | 查询数据库返回 | - |
| AI生成 | 显示Loading状态，展示预览卡片 | 组装数据，调用Python，保存结果 | 调LLM生成内容，返回JSON |
| 数据回填 | 更新本地状态，刷新主页面 | 保存到数据库，返回成功 | - |
| 指令修改 | 发送聊天消息 | 转发给Python，更新数据 | 理解指令，返回修改后的数据 |
| 登录认证 | 携带Token | 验证Token，返回401 if过期 | - |
| 检查功能 | 展示检查报告 | 调用Python执行检查逻辑 | 执行方案书完整性检查 |

**错误码约定**：
- 200: 成功
- 400: 请求参数错误
- 401: 未登录/Token过期 → 前端跳转登录页
- 403: 无权限（非项目经理）
- 500: 服务器错误
- 503: AI服务不可用 → 前端提示"AI服务暂时不可用"
